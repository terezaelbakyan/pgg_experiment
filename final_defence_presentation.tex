% SLIDE 2 — Topic & Research Question
%----------------------------------------------------------
\begin{frame}{Topic \& Research Question}

\begin{block}{Research Question}
\small
Does pre-play cheap talk improve coordination in a threshold public goods game,
and does the richness of the communication channel determine the size of the effect?
\end{block}

\vspace{0.5em}
\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \textbf{The problem}
    \begin{itemize}\small
      \item A provision point is introduced: the public good is provided only when combined
            contributions reach a minimum level
      \item Free-riding incentives exist alongside a coordination problem
      \item An individual player's contribution depends on the belief that others will
            contribute enough for it to matter
      \item Without a communication device, players may shift toward the inferior equilibrium
    \end{itemize}
  \end{column}
  \begin{column}{0.44\textwidth}
    \textbf{Three treatments}
    \begin{itemize}\small
      \item \cg{Control} --- no communication before each contribution decision
      \item \bo{Binary} --- numerical intention signal (0--10, non-binding)
      \item \cb{Chat} --- 90-second unrestricted anonymous text chat
    \end{itemize}
    \vspace{0.4em}
    \small Everything else is the same across treatments.
    Differences in outcomes are attributed to communication alone.
  \end{column}
\end{columns}

\end{frame}

%----------------------------------------------------------
% SLIDE 3 — Prior Work: P&R 1991 + PRR 2016
%----------------------------------------------------------
\begin{frame}{Prior Work: Two Key Papers}

\begin{columns}[T]
  \begin{column}{0.48\textwidth}
    \begin{block}{Palfrey \& Rosenthal (1991)}
      \small
      Threshold PGG with \textbf{private information} about contribution costs.
      Implemented a binary communication mechanism.\\[0.4em]
      Demonstrated that perfect coordination is not Bayesian incentive compatible;
      weak free-riding incentives are retained.
      Proved the existence of communication equilibria that improve efficiency.\\[0.4em]
      \textbf{Empirical finding:} Binary communication does \emph{not} produce more efficient outcomes
      compared with the no-communication baseline.
    \end{block}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{block}{Palfrey, Rosenthal \& Roy (2016)}
      \small
      Developed a Bayesian mechanism design to estimate efficiency bounds under three communication structures:
      binary, numerical cost reports, and unrestricted text chat.\\[0.4em]
      Efficiency bounds depend on the richness of the message space
      and the distribution of private information.\\[0.4em]
      \textbf{Finding:} Significant efficiency gains arise with unrestricted chat \emph{only}.
      Binary and numerical signals have little to no effect.
    \end{block}
  \end{column}
\end{columns}

\vspace{0.4em}
\begin{block}{Relationship to PRR (2016)}
  \small
  PRR (2016) is the \textbf{primary base} of this study both theoretically and empirically.
  The present experiment replicates their numerical and chat treatments
  in a different subject pool and contribution setting.
\end{block}

\end{frame}

% SLIDE 4 — Their Game → Your Game
%----------------------------------------------------------
\begin{frame}{Their Game $\longrightarrow$ Your Game}

\begin{columns}[T]
  \begin{column}{0.46\textwidth}
    \textbf{PRR (2016) setup}
    \begin{itemize}\small
      \item Threshold public goods game
      \item \textbf{Private information} about contribution costs and endowment types
      \item Three communication structures: binary, numerical, unrestricted chat
      \item Found significant efficiency gains from unrestricted chat in a structurally similar game
    \end{itemize}
  \end{column}
  \begin{column}{0.50\textwidth}
    \textbf{This study}
    \begin{itemize}\small
      \item Same threshold public goods game structure
      \item \textbf{Complete information}: all parameters are disclosed to every player\\
            ($e=10$, $T=15$, $m=2$ are common knowledge)
      \item Three treatments: \cg{Control}, \bo{Binary}, \cb{Chat}
      \item Subject pool: AUA undergraduate students, Armenia
      \item 90-second chat window (PRR used 60 seconds) --- extra 30 seconds to offset the
            language barrier; participants' native language is Armenian
    \end{itemize}
  \end{column}
\end{columns}

\vspace{0.4em}
\begin{block}{}
  \small
  Complete information simplifies the environment but introduces a \emph{multiplicity of equilibria}.
  The implications of this for the research question are central to the study.
\end{block}

\end{frame}

%----------------------------------------------------------
% SLIDE 5 — Game Structure
%----------------------------------------------------------
\begin{frame}{Game Structure}

\begin{columns}[T]
  \begin{column}{0.45\textwidth}
    \textbf{Parameters}
    \begin{itemize}\small
      \item $N = 3$ players per group
      \item Endowment $e = 10$ coins per round (do not accumulate)
      \item Threshold $T = 15$ coins
      \item Multiplier $m = 2$
      \item $c_p \in \{0, 1, \ldots, 10\}$; contributions are non-refundable
    \end{itemize}

    \vspace{0.5em}
    \textbf{Payoff function}
    \[
      \pi_p =
      \begin{cases}
        (e - c_p) + \dfrac{2\displaystyle\sum c_j}{3}
          & \text{if } \textstyle\sum c_j \geq 15 \\[8pt]
        e - c_p
          & \text{if } \textstyle\sum c_j < 15
      \end{cases}
    \]
  \end{column}

  \begin{column}{0.51\textwidth}
    \textbf{Two focal Nash equilibria}

    \vspace{0.2em}
    \begin{block}{\cg{Free-riding: $(0,\,0,\,0)$ $\to$ $\pi = 10$}}
      \small Contributing zero is a dominant response when others contribute zero.
      Threshold is never met regardless of player $p$'s decision.
    \end{block}

    \vspace{0.2em}
    \begin{block}{\cb{Coordinated: $(5,\,5,\,5)$ $\to$ $\pi = 15$}}
      \small $c_p = 5$ maximises individual payoff given others contribute 5.\\
      $\pi_p = (10{-}5) + \tfrac{2 \times 15}{3} = 5 + 10 = 15$.
      No player has an incentive to deviate.
    \end{block}

    \vspace{0.3em}
    \small
    \textbf{92} pure-strategy Nash equilibria in total;
    \textbf{74} satisfy trembling-hand perfection (THPE);
    \textbf{18} are not THPE (any equilibrium where some $c_p = 10$).

    \vspace{0.4em}
    $\Rightarrow$ Multiple equilibria; common knowledge of rationality alone cannot select among them.
    We study \textbf{3 treatments}.
  \end{column}
\end{columns}

\end{frame}

%----------------------------------------------------------
% SLIDE 6 — Three Treatments (one by one)
%----------------------------------------------------------
\begin{frame}{Three Treatments}

\begin{columns}[T]
  \begin{column}{0.31\textwidth}
    \begin{block}{\cg{Control}}
      \small
      \textbf{One stage:}\\[0.3em]
      A decision page is displayed.
      Participants enter a contribution
      from 0 to 10 coins.\\[0.3em]
      No prior communication of any kind.\\[0.3em]
      \textbf{Baseline treatment.}
    \end{block}
  \end{column}

  \begin{column}{0.31\textwidth}
    \begin{block}{\bo{Binary}}
      \small
      \textbf{Two stages:}\\[0.3em]
      \textit{Communication:} Each player states a numerical intention (0--10).
      All intentions are simultaneously revealed to the group.\\[0.3em]
      \textit{Contribution:} Players enter their contribution (0--10 coins).\\[0.3em]
      Intentions are \textbf{non-binding}.
    \end{block}
  \end{column}

  \begin{column}{0.31\textwidth}
    \begin{block}{\cb{Chat}}
      \small
      \textbf{Two stages:}\\[0.3em]
      \textit{Communication:} 90-second unrestricted, anonymous group chat.
      Rules: relevant messages only; no identity reveal; no offensive language; no nonsense.\\[0.3em]
      \textit{Contribution:} Players enter their contribution (0--10 coins).\\[0.3em]
      Communication is \textbf{non-binding}.
    \end{block}
  \end{column}
\end{columns}

\vspace{0.45em}
\small
\textbf{Sample:} 135 AUA undergraduates \textbullet\;
4 sessions (36 + 36 + 36 + 27) \textbullet\;
45 per treatment \textbullet\;
13 rounds (number not disclosed) \textbullet\;
1,755 observations.\\
Random rematching within treatment each round to prevent reputation effects.
oTree (Chen, Schonger \& Wickens, 2016). \; IRB: AUA-2026-016.

\end{frame}

\begin{frame}{Hypotheses \& Results: Contributions (TH1 \& TH2)}

\begin{columns}[T]
  \begin{column}{0.50\textwidth}
    \textbf{Hypotheses}\\[0.3em]
    \small
    \textbf{TH1:} \cb{Chat} increases average individual contributions
    relative to \cg{Control}.\\[0.4em]
    \textbf{TH2:} \bo{Binary} decreases average individual contributions
    relative to \cg{Control}.

\begin{frame}{Results \& Limitations}

\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \small\textbf{Summary (5 hypotheses)}\\[0.3em]
    \footnotesize
    \setlength{\tabcolsep}{3pt}
    \renewcommand{\arraystretch}{1.30}
    \begin{tabular}{@{}p{0.7cm}p{4.0cm}c@{}}
      \toprule
      \textbf{Hyp.} & \textbf{Prediction} & \\
      \midrule
      \textbf{TH1} & \cb{Chat} $\uparrow$ contributions & \yes \\
                   & \tiny $\hat\beta{=}+0.448$,\; $p{=}0.009$ & \\[2pt]
      \textbf{TH2} & \bo{Binary} $\downarrow$ contributions & \yes \\
                   & \tiny $\hat\beta{=}-0.561$,\; $p{=}0.001$ & \\[2pt]
      \textbf{GH3} & \bo{Binary} $\downarrow$ payoffs & \yes \\
                   & \tiny $\hat\beta{=}-1.474^{***}$,\; $p{<}0.001$ & \\[2pt]
      \textbf{GH4} & \cb{Chat} $\uparrow$ payoffs & \no \\
                   & \tiny $\hat\beta{=}+0.249$,\; $p{=}0.398$ & \\[2pt]
      \textbf{GH5} & Intentions $>$ contributions & \no \\
                   & \tiny Reversed: $\Delta{=}+0.857^{***}$ & \\
      \bottomrule
    \end{tabular}

    \vspace{0.35em}
    \small
    \textbf{Central finding:} The effectiveness of communication critically depends on its form.
    A single number may just as easily anchor the group on failure as on success.
    No evidence of learning across rounds was found.
  \end{column}

  \begin{column}{0.44\textwidth}
    \small\textbf{Limitations}
    \begin{itemize}\footnotesize
      \item No monetary compensation (course participation credit only) --- reduces external validity and weakens incentive compatibility
      \item A broader and more diverse sample would strengthen confidence in results
      \item Only 13 rounds within a 30--50 minute constraint --- longer sessions would reveal more patterns
      \item AUA labs are not designed for behavioural experiments
    \end{itemize}

    \vspace{0.3em}
    \small\textbf{Future research}
    \begin{itemize}\footnotesize
      \item Identify the minimum communication richness required for a coordination gain to emerge
    \end{itemize}

    \vspace{0.25em}
    \begin{block}{}
      \small
      Unrestricted chat improves coordination;
      binary signalling can make outcomes worse than no communication at all.
    \end{block}
  \end{column}
\end{columns}
